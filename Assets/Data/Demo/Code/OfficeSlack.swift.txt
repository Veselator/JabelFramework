// OfficeSlack.swift
// The office chat for monkeys: channels, mentions, reactions and a bot
// that reminds everybody to drink water and stop typing in caps.

import Foundation

enum Reaction: String, CaseIterable {
    case banana = ":banana:"
    case thumbsUp = ":+1:"
    case monkey = ":see_no_evil:"
    case fire = ":fire:"
    case coffee = ":coffee:"
}

struct Message: Identifiable {
    let id: Int
    let author: String
    let text: String
    let sentAt: Date
    var reactions: [Reaction: Set<String>] = [:]

    var mentions: [String] {
        text.split(separator: " ")
            .filter { $0.hasPrefix("@") }
            .map { String($0.dropFirst()).trimmingCharacters(in: .punctuationCharacters) }
    }

    var isShouting: Bool {
        let letters = text.filter { $0.isLetter }
        return letters.count > 8 && letters.allSatisfy { $0.isUppercase }
    }
}

final class Channel {
    let name: String
    private(set) var members: Set<String> = []
    private(set) var messages: [Message] = []
    private var nextId = 1

    init(name: String) {
        self.name = name
    }

    func join(_ monkey: String) {
        members.insert(monkey)
    }

    @discardableResult
    func post(_ text: String, by author: String) -> Message? {
        guard members.contains(author) else { return nil }
        let message = Message(id: nextId, author: author, text: text, sentAt: Date())
        nextId += 1
        messages.append(message)
        return message
    }

    func react(_ reaction: Reaction, to messageId: Int, by monkey: String) {
        guard let index = messages.firstIndex(where: { $0.id == messageId }) else { return }
        messages[index].reactions[reaction, default: []].insert(monkey)
    }

    func unread(for monkey: String, since date: Date) -> [Message] {
        messages.filter { $0.sentAt > date && $0.author != monkey }
    }

    func mostPopular() -> Message? {
        messages.max { a, b in
            a.reactions.values.reduce(0) { $0 + $1.count } < b.reactions.values.reduce(0) { $0 + $1.count }
        }
    }
}

protocol Bot {
    var name: String { get }
    func respond(to message: Message, in channel: Channel) -> String?
}

struct WellnessBot: Bot {
    let name = "wellness-bot"

    func respond(to message: Message, in channel: Channel) -> String? {
        if message.isShouting {
            return "@\(message.author) please stop typing in caps, the keyboard is fine."
        }
        if message.text.lowercased().contains("tired") {
            return "@\(message.author) have a banana and a glass of water."
        }
        return nil
    }
}

struct StandupBot: Bot {
    let name = "standup-bot"

    func respond(to message: Message, in channel: Channel) -> String? {
        guard message.text.lowercased().hasPrefix("/standup") else { return nil }
        let names = channel.members.sorted().joined(separator: ", ")
        return "Stand-up in 5 minutes: \(names). Bring bananas."
    }
}

final class Workspace {
    private(set) var channels: [String: Channel] = [:]
    private var bots: [Bot] = []

    func channel(_ name: String) -> Channel {
        if let existing = channels[name] { return existing }
        let created = Channel(name: name)
        channels[name] = created
        return created
    }

    func install(_ bot: Bot) {
        bots.append(bot)
    }

    func send(_ text: String, from monkey: String, to channelName: String) {
        let channel = channel(channelName)
        channel.join(monkey)
        guard let message = channel.post(text, by: monkey) else { return }
        for mention in message.mentions where !channel.members.contains(mention) {
            print("[notify] \(mention) was mentioned in #\(channelName) but is not a member")
        }
        for bot in bots {
            if let reply = bot.respond(to: message, in: channel) {
                channel.join(bot.name)
                channel.post(reply, by: bot.name)
            }
        }
    }

    func transcript(of channelName: String) -> String {
        guard let channel = channels[channelName] else { return "(no such channel)" }
        return channel.messages.map { message in
            let reactions = message.reactions
                .map { "\($0.key.rawValue)\($0.value.count)" }
                .sorted()
                .joined(separator: " ")
            return "\(message.author): \(message.text) \(reactions)"
        }.joined(separator: "\n")
    }
}

let workspace = Workspace()
workspace.install(WellnessBot())
workspace.install(StandupBot())

workspace.send("good morning jungle", from: "Bobo", to: "general")
workspace.send("WHO ATE MY BANANA FROM THE FRIDGE", from: "Koko", to: "general")
workspace.send("@Koko it was @Professor", from: "Momo", to: "general")
workspace.send("I am so tired of semicolons", from: "Chip", to: "general")
workspace.send("/standup", from: "Bobo", to: "general")

let general = workspace.channel("general")
if let first = general.messages.first {
    for monkey in ["Koko", "Momo", "Chip"] {
        general.react(.banana, to: first.id, by: monkey)
    }
    general.react(.coffee, to: first.id, by: "Bobo")
}

print(workspace.transcript(of: "general"))
if let top = general.mostPopular() {
    print("Most loved message: \"\(top.text)\" by \(top.author)")
}
